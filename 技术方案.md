**Intent 调用代码示例：**

```dart
// Flutter 端
class ExternalToolService {
  static const _channel = MethodChannel('com.misside/external_tools');
  
  /// 启动 MT Manager 进行反编译
  Future<void> decompileWithMT(String apkPath) async {
    final uri = Uri.file(apkPath);
    await _channel.invokeMethod('sendIntent', {
      'action': 'android.intent.action.VIEW',
      'type': 'application/vnd.android.package-archive',
      'data': uri.toString(),
      'package': 'com.mt.manager',
      'flags': ['FLAG_ACTIVITY_NEW_TASK'],
    });
  }
  
  /// 通过 Termux 执行 shell 命令
  Future<String> executeInTermux(String command) async {
    final result = await _channel.invokeMethod('termuxExec', {
      'command': command,
      'workdir': await _getWorkDirectory(),
    });
    return result as String;
  }
}
```

2. **创建 MethodChannel：**
```kotlin
// Android 端: JadxEnginePlugin.kt
class JadxEnginePlugin: MethodChannel.MethodCallHandler {
    private val jadxCtx = JadxCtxImpl()
    
    override fun onMethodCall(call: MethodCall, result: Result) {
        when (call.method) {
            "decompileDex" -> decompileDex(call, result)
            "decodeResources" -> decodeResources(call, result)
            "getVersion" -> result.success("1.4.7")
        }
    }
    
    private fun decompileDex(call: MethodCall, result: Result) {
        val dexPath = call.argument<String>("dexPath")!!
        val outputDir = call.argument<String>("outputDir")!!
        
        thread {
            try {
                val args = DexInputArgs().apply {
                    addFile(File(dexPath))
                }
                Jadx.run(args)
                result.success(true)
            } catch (e: Exception) {
                result.error("DECOMPILE_ERROR", e.message, null)
            }
        }
    }
}
```

#### 算法对比分析

| 算法 | 复杂度 | 优点 | 缺点 | 适用场景 |
|------|--------|------|------|----------|
| **Myers** | O(ND) | 速度快、实现成熟 | 可能匹配无意义行 | 通用文本、简单变更 |
| **Patience** | O(N log N) | 匹配有意义唯一行 | 慢于 Myers | 代码重构、函数重排 |
| **Histogram** | O(N) | 结合两者优点 | 复杂 | Git 默认替选 |
| **Semantic** | 依赖 AST | 理解代码结构 | 需完整 Parser | 语言特定优化 |

**推荐方案：**
- **Smali 对比**：Myers Diff（轻量、快速）
- **Java 对比**：Patience Diff（重构友好）
- **Dex 对比**：自定义二进制 Diff

### 3.2 三种文件类型对比实现

#### 3.2.1 Smali 对比实现

```dart
class SmaliDiffEngine implements DiffEngine {
  @override
  DiffResult compare(String left, String right, {DiffOptions? options}) {
    // Smali 特点：基于行的Dalvik字节码
    // 1. 先按方法/类结构分割
    final leftMethods = _parseSmaliMethods(left);
    final rightMethods = _parseSmaliMethods(right);
    
    // 2. 方法级别对齐
    final aligned = _alignMethods(leftMethods, rightMethods);
    
    // 3. 行级别 Myers Diff
    final result = DiffResult();
    for (final pair in aligned) {
      final diff = _myersDiff(pair.leftLines, pair.rightLines);
      result.addChunk(DiffChunk(
        type: _classifyChange(diff),
        leftLines: pair.leftLines,
        rightLines: pair.rightLines,
        context: _extractContext(pair),
      ));
    }
    
    return result;
  }
  
  /// Smali 方法解析
  List<SmaliMethod> _parseSmaliMethods(String content) {
    final methods = <SmaliMethod>[];
    final regex = RegExp(r'\.method\s+(.*?)\s+(.*?)\((.*?)\)(.*?)\n(.*?)\.end method', multiLine: true);
    
    for (final match in regex.allMatches(content)) {
      methods.add(SmaliMethod(
        accessFlags: match.group(1)!,
        name: match.group(2)!,
        params: match.group(3)!,
        returnType: match.group(4)!,
        body: match.group(5)!,
        lineStart: match.start,
        lineEnd: match.end,
      ));
    }
    return methods;
  }
  
  /// Smali 特定变化分类
  ChangeType _classifyChange(DiffChunk chunk) {
    // 检测寄存器数量变化、方法签名变化等 Smali 特定模式
    if (_hasRegisterCountChange(chunk)) return ChangeType.SEMANTIC_CHANGE;
    if (_hasMethodSignatureChange(chunk)) return ChangeType.SIGNATURE_CHANGE;
    return ChangeType.NORMAL;
  }
}
```

```dart
  DiffResult compare(String left, String right, {DiffOptions? options}) {
    // 1. AST 级别 Patience Diff（用于高层面结构）
    final leftAst = _parseJavaAst(left);
    final rightAst = _parseJavaAst(right);
    final astDiff = _patienceDiff(leftAst.methods, rightAst.methods);
    
    // 2. 行级别细粒度 Diff
    final result = DiffResult();
    for (final node in astDiff.changes) {
      if (node.type == ChangeType.MODIFY) {
        // 方法体内使用 Myers Diff
        final methodDiff = _myersDiff(
          _extractMethodBody(leftAst, node.leftId),
          _extractMethodBody(rightAst, node.rightId),
        );
        result.addChunk(methodDiff);
      } else {
        // 整体添加/删除
        result.addChunk(_createChunk(node));
      }
    }
    
    return result;
  }
  
  /// Patience Diff 实现
  List<PatienceChange> _patienceDiff(List<MethodNode> left, List<MethodNode> right) {
    // 1. 找唯一公共元素（方法签名）
    final uniqueInLeft = left.where((m) => 
      right.where((r) => r.signature == m.signature).isEmpty).toList();
    final uniqueInRight = right.where((m) => 
      left.where((l) => l.signature == m.signature).isEmpty).toList();
    final common = left.where((m) => 
      right.any((r) => r.signature == m.signature)).toList();
    
    // 2. LCS 匹配公共方法
    final lcs = _longestCommonSubsequence(common, right);
    
    // 3. 组合结果
    return _buildDiffSequence(uniqueInLeft, lcs, uniqueInRight);
  }
}
```

#### 3.2.3 Dex 对比实现

```dart
class DexDiffEngine implements DiffEngine {
  @override
  DiffResult compare(Uint8List leftDex, Uint8List rightDex, {DiffOptions? options}) {
    // Dex 是二进制格式，需要解析后再对比
    
    // 1. 解析 Dex 结构
    final leftParsed = DexParser.parse(leftDex);
    final rightParsed = DexParser.parse(rightDex);
    
    // 2. 提取可对比元素
    final leftClasses = leftParsed.classes;
    final rightClasses = rightParsed.classes;
    
    // 3. 类级别差异分析
    final result = DiffResult();
    final aligned = _alignClasses(leftClasses, rightClasses);
    
    for (final pair in aligned) {
      final diff = _compareClass(pair.left, pair.right);
      result.addChunk(diff);
    }
    
    // 4. 生成人类可读报告
    return _generateReadableReport(result);
  }
  
  /// Dex 类对比
  DiffChunk _compareClass(DexClass? left, DexClass? right) {
    if (left == null && right != null) {
      return DiffChunk(type: ChangeType.ADD, 
        description: 'New class: ${right.name}');
    }
    if (left != null && right == null) {
      return DiffChunk(type: ChangeType.DELETE,
        description: 'Removed class: ${left.name}');
    }
    
    // 方法对比
    final methodDiff = _compareMethods(left!.methods, right!.methods);
    return DiffChunk(
      type: ChangeType.MODIFY,
      subChanges: methodDiff,
      description: 'Modified class: ${left.name}',
    );
  }
}
```

```dart
class DualEditController {
  // 左侧编辑器控制器
  late final CodeController leftController;
  // 右侧编辑器控制器
  late final CodeController rightController;
  
  // 同步状态
  final _syncState = DualSyncState();
  // 是否正在同步中（防止死循环）
  bool _isSyncing = false;
  
  // 同步滚动
  bool syncScrollEnabled = true;
  // 同步策略
  ConflictStrategy conflictStrategy = ConflictStrategy.MANUAL;
  
  void initialize() {
    // 监听左侧编辑
    leftController.addListener(_onLeftChanged);
    // 监听右侧编辑
    rightController.addListener(_onRightChanged);
    
    // 监听滚动
    leftController.addOnScrollListener(_onLeftScroll);
    rightController.addOnScrollListener(_onRightScroll);
  }
  
  void _onLeftChanged() {
    if (_isSyncing) return;
    _isSyncing = true;
    
    try {
      final delta = _calculateDelta(
        oldContent: _syncState.lastLeftContent,
        newContent: leftController.text,
      );
      
      // 检测冲突
      final conflicts = _detectConflicts(delta, rightController.text);
      
      if (conflicts.isNotEmpty) {
        _handleConflicts(conflicts, ConflictSide.LEFT);
      } else {
        // 无冲突，应用到右侧
        _applyDeltaToRight(delta);
      }
      
      _syncState.lastLeftContent = leftController.text;
    } finally {
      _isSyncing = false;
    }
  }
  
  void _onRightChanged() {
    if (_isSyncing) return;
    _isSyncing = true;
    
    try {
      final delta = _calculateDelta(
        oldContent: _syncState.lastRightContent,
        newContent: rightController.text,
      );
      
      final conflicts = _detectConflicts(delta, leftController.text);
      
      if (conflicts.isNotEmpty) {
        _handleConflicts(conflicts, ConflictSide.RIGHT);
      } else {
        _applyDeltaToLeft(delta);
      }
      
      _syncState.lastRightContent = rightController.text;
    } finally {
      _isSyncing = false;
    }
  }
  
  List<EditConflict> _detectConflicts(TextDelta delta, String otherContent) {
    final conflicts = <EditConflict>[];
    
    for (final edit in delta.changes) {
      // 检测是否与右侧编辑重叠
      if (_overlapsWithOtherEdit(edit, otherContent)) {
        conflicts.add(EditConflict(
          type: ConflictType.OVERLAPPING_EDIT,
          leftRange: edit.range,
          rightContent: _getAffectedContent(otherContent, edit.range),
          severity: ConflictSeverity.MEDIUM,
        ));
      }
    }
    
    return conflicts;
  }
  
  void _handleConflicts(List<EditConflict> conflicts, ConflictSide source) {
    switch (conflictStrategy) {
      case ConflictStrategy.LAST_WRITE_WINS:
        // 最后写入优先，无需处理
        break;
        
      case ConflictStrategy.LEFT_PRIORITY:
        if (source == ConflictSide.RIGHT) {
          // 丢弃右侧修改
          rightController.text = _syncState.lastRightContent;
        }
        break;
        
      case ConflictStrategy.RIGHT_PRIORITY:
        if (source == ConflictSide.LEFT) {
          // 丢弃左侧修改
          leftController.text = _syncState.lastLeftContent;
        }
        break;
        
      case ConflictStrategy.MANUAL:
        // 弹出冲突解决对话框
        _showConflictDialog(conflicts);
        break;
        
      case ConflictStrategy.MERGE:
        // 智能合并
        _smartMerge(conflicts);
        break;
    }
  }
  
  // 同步滚动处理
  void _onLeftScroll(double scrollOffset) {
    if (!syncScrollEnabled || _isSyncing) return;
    
    _isSyncing = true;
    try {
      // 计算对应的右侧滚动位置（考虑行数差异）
      final rightOffset = _calculateCorrespondingOffset(
        scrollOffset,
        leftController.text,
        rightController.text,
      );
      rightController.scrollTo(rightOffset);
    } finally {
      _isSyncing = false;
    }
  }
}
```

```dart
class DiffGestureController {
  // 手势冲突仲裁
  final GestureArenaManager _arenaManager = GestureArenaManager();
  
  // 当前手势模式
  GestureMode _currentMode = GestureMode.edit;
  
  // 编辑器控制器
  final CodeController leftController;
  final CodeController rightController;
  
  // 差异导航
  final DiffNavigator _diffNavigator;
  
  // 同步滚动控制器
  final ScrollSyncController _scrollSync;
  
  DiffGestureController({
    required this.leftController,
    required this.rightController,
    required DiffNavigator diffNavigator,
  }) : _diffNavigator = diffNavigator,
       _scrollSync = ScrollSyncController() {
    _initGestureDetectors();
  }
  
  void _initGestureDetectors() {
    // 主编辑器手势
    final editorGestureDetector = GestureDetector(
      // 水平拖动 → 差异导航
      onHorizontalDragUpdate: _onHorizontalDrag,
      onHorizontalDragEnd: _onHorizontalDragEnd,
      
      // 垂直拖动 → 滚动
      onVerticalDragUpdate: _onVerticalScroll,
      
      // 双指捏合 → 缩放字体
      onScaleUpdate: _onPinchToZoom,
      
      // 长按 → 选择模式
      onLongPressStart: _onLongPress,
      onLongPressMoveUpdate: _onLongPressMove,
      onLongPressEnd: _onLongPressEnd,
    );
    
    // 三指手势 → 合并操作
    final threeFingerDetector = GestureDetector(
      onScaleStart: _onThreeFingerStart,
      onScaleUpdate: _onThreeFingerUpdate,
      onScaleEnd: _onThreeFingerEnd,
    );
  }
  
  // 水平滑动手势处理（差异导航）
  Offset? _dragStart;
  double _dragThreshold = 30.0;  // 滑动阈值
  int _currentDiffIndex = 0;
  
  void _onHorizontalDrag(DragUpdateDetails details) {
    if (_currentMode != GestureMode.edit) return;
    
    _dragStart ??= details.globalPosition;
    final delta = details.globalPosition - _dragStart!;
    
    // 水平滑动超过阈值
    if (delta.dx.abs() > _dragThreshold && delta.dy.abs() < _dragThreshold) {
      _currentMode = GestureMode.diffNavigation;
      
      // 显示差异导航指示器
      _showDiffNavigatorIndicator(
        direction: delta.dx > 0 ? NavigationDirection.next : NavigationDirection.prev,
      );
    }
  }
  
  void _onHorizontalDragEnd(DragEndDetails details) {
    if (_currentMode == GestureMode.diffNavigation) {
      final delta = details.velocity.pixelsPerSecond;
      
      if (delta.dx > 0) {
        // 向右滑 → 下一个差异
        _diffNavigator.navigateToNext();
      } else {
        // 向左滑 → 上一个差异
        _diffNavigator.navigateToPrev();
      }
    }
    
    _dragStart = null;
    _currentMode = GestureMode.edit;
    _hideDiffNavigatorIndicator();
  }
  
  // 同步滚动处理
  void _onVerticalScroll(DragUpdateDetails details) {
    if (!syncScrollEnabled) return;
    
    _scrollSync.syncScroll(
      source: _getActiveEditor(),
      delta: details.delta,
    );
  }
  
  // 三指合并手势
  Direction? _threeFingerDirection;
  
  void _onThreeFingerStart(ScaleStartDetails details) {
    if (details.pointerCount == 3) {
      _threeFingerDirection = null;
    }
  }
  
  void _onThreeFingerUpdate(ScaleUpdateDetails details) {
    if (details.pointerCount != 3) return;
    
    final dx = details.focalPointDelta.dx;
    
    if (_threeFingerDirection == null && dx.abs() > 20) {
      _threeFingerDirection = dx > 0 ? Direction.right : Direction.left;
      _showMergeIndicator(direction: _threeFingerDirection!);
    }
    
    if (_threeFingerDirection != null) {
      // 更新合并预览
      _updateMergePreview(direction: _threeFingerDirection!);
    }
  }
  
  void _onThreeFingerEnd(ScaleEndDetails details) {
    if (_threeFingerDirection != null) {
      // 执行合并
      if (_threeFingerDirection == Direction.left) {
        _mergeFromLeft();
      } else {
        _mergeFromRight();
      }
    }
    
    _threeFingerDirection = null;
    _hideMergeIndicator();
  }
  
  // 摇一摇撤销
  void _setupShakeToUndo() {
    ShakeDetector.create(
      onShake: () {
        if (_currentMode == GestureMode.edit) {
          _undoLastEdit();
          _showUndoFeedback();
        }
      },
      shakeThreshold: 15,
      shakeCountResetTime: 500,
    );
  }
}
```

```dart
class LazyFileLoader {
  static const int pageSize = 1000;
  static const int preloadPages = 2;
  
  final _cache = LRU cache<String, List<String>>(maxSize: 10);
  
  Future<List<String>> loadLines(String filePath, int startLine, int count) async {
    final cacheKey = '$filePath:$startLine';
    
    if (_cache.containsKey(cacheKey)) {
      return _cache.get(cacheKey)!;
    }
    
    final file = File(filePath);
    final lines = await file
        .openRead()
        .transform(utf8.decoder)
        .transform(const LineSplitter())
        .skip(startLine)
        .take(count)
        .toList();
    
    _cache.put(cacheKey, lines);
    return lines;
  }
  
  /// 预加载相邻页面
  Future<void> preloadAdjacent(String filePath, int currentLine) async {
    final pageStart = (currentLine ~/ pageSize) * pageSize;
    
    // 预加载前一个页面
    if (pageStart > 0) {
      loadLines(filePath, pageStart - pageSize, pageSize);
    }
    
    // 预加载后一个页面
    loadLines(filePath, pageStart + pageSize, pageSize);
  }
}
```

```kotlin
    private val executor = Executors.newFixedThreadPool(2)
    
    override fun onMethodCall(call: MethodCall, result: Result) {
        when (call.method) {

                val options = call.argument<Map<String, Any>>("options")
                
                // 后台执行
                executor.execute {
                    try {
                        val outputDir = decompile(apkPath, options)
                        
                        // 序列化结果
                        val fileList = listFiles(outputDir)
                        val resultMap = mapOf(
                            "success" to true,
                            "outputDir" to outputDir,
                            "files" to fileList.map { it.absolutePath },
                            "stats" to getStats(outputDir)
                        )
                        
                        // 回到主线程返回
                        Handler(Looper.getMainLooper()).post {
                            result.success(resultMap)
                        }
                    } catch (e: Exception) {
                        Handler(Looper.getMainLooper()).post {
                            result.error("DECOMPILE_ERROR", e.message, e.stackTraceToString())
                        }
                    }
                }
            }
        }
    }
}
```

#### 难点 4：Diff 算法在移动端的优化

**问题描述：**
- 传统 Myers Diff 对大文件可能 O(N²)
- Patience Diff 需要额外的数据结构
- 移动端 CPU 性能有限

**解决方案：**

```dart
class OptimizedDiffEngine {
  /// 自适应 Diff 算法选择
  DiffResult compute(String left, String right, FileType type) {
    final leftLines = left.split('\n');
    final rightLines = right.split('\n');
    
    // 文件大小阈值
    const smallFileThreshold = 500;    // 500 行
    const mediumFileThreshold = 5000;   // 5000 行
    
    if (leftLines.length + rightLines.length < smallFileThreshold) {
      // 小文件：使用完整 Myers Diff
      return _myersDiff(leftLines, rightLines);
    } else if (leftLines.length + rightLines.length < mediumFileThreshold) {
      // 中文件：使用空间优化版 Myers
      return _spaceOptimizedDiff(leftLines, rightLines);
    } else {
      // 大文件：分块处理 + 并行计算
      return _chunkedDiff(leftLines, rightLines);
    }
  }
  
  /// 分块 Diff（核心优化）
  DiffResult _chunkedDiff(List<String> left, List<String> right) {
    const chunkSize = 500;
    
    // 1. 快速哈希找锚点
    final leftAnchors = _findAnchors(left, chunkSize);
    final rightAnchors = _findAnchors(right, chunkSize);
    
    // 2. 锚点对齐
    final alignedChunks = _alignChunks(leftAnchors, rightAnchors);
    
    // 3. 对齐区域内部使用 Myers Diff
    // 4. 未对齐区域使用启发式合并
    return _mergeResults(alignedChunks);
  }
  
  /// 找锚点（长公共子串）
  List<int> _findAnchors(List<String> lines, int interval) {
    final anchors = <int>[];
    for (int i = 0; i < lines.length; i += interval) {
      anchors.add(_hash(lines[i]));
    }
    return anchors;
  }
}
```

#### 难点 5：Android 版本兼容性

**问题描述：**
- Android 5.0-14+ 版本差异
- 分区存储 (Scoped Storage) 对文件访问的影响
- 不同厂商的权限管理差异

**解决方案：**

```dart
class AndroidCompatibilityService {
  /// 检测并适配不同 Android 版本
  Future<void> initialize() async {
    final sdkInt = await _getSdkInt();
    
    if (sdkInt >= 30) {
      // Android 11+：使用 Scoped Storage
      await _initScopedStorage();
    } else if (sdkInt >= 21) {
      // Android 5-10：传统存储
      await _initLegacyStorage();
    }
    
    // 权限适配
    if (sdkInt >= 33) {
      // Android 13+：READ_MEDIA_* 权限
      await _requestMediaPermissions();
    } else {
      // 传统存储权限
      await _requestStoragePermissions();
    }
  }
  
  /// 外部工具调用兼容性
  Future<bool> canInvokeExternalTool(String packageName) async {
    final sdkInt = await _getSdkInt();
    
    if (sdkInt >= 30) {
      // Android 11+：包可见性限制
      return await _checkPackageVisibility(packageName);
    }
    
    return true;
  }
}
```
